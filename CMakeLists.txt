cmake_minimum_required(VERSION 3.16)
project(bibilabu_ps3 VERSION 1.0.0 LANGUAGES CXX)
option(PS3_PORT_BUILD_TESTS "Build host tests for PS3 stream policy" OFF)
add_library(bibilabu_ps3_policy STATIC src/stream_policy.cpp)
target_include_directories(bibilabu_ps3_policy PUBLIC include)
target_compile_features(bibilabu_ps3_policy PUBLIC cxx_std_17)
target_compile_options(bibilabu_ps3_policy PRIVATE -Wall -Wextra -Werror)
if(PS3_PORT_BUILD_TESTS)
    enable_testing()
    add_executable(ps3_policy_tests tests/policy_tests.cpp)
    target_link_libraries(ps3_policy_tests PRIVATE bibilabu_ps3_policy)
    add_test(NAME ps3_stream_policy COMMAND ps3_policy_tests)
endif()
